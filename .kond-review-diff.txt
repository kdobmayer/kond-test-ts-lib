+import { ValidationIssue, ValidationResult, ValidationSeverity } from '../types';
+
+export class ValidationReport {
+  private _results: ValidationResult[] = [];
+  private _issues: ValidationIssue[] = [];
+
+  addResult(result: ValidationResult): void {
+    this._results.push(result);
+    for (const issue of result.issues) {
+      this._issues.push(issue);
+    }
+  }
+
+  addIssue(issue: ValidationIssue): void {
+    this._issues.push(issue);
+  }
+
+  getSummary(): {
+    totalResults: number;
+    totalIssues: number;
+    errorCount: number;
+    warningCount: number;
+    infoCount: number;
+    valid: boolean;
+  } {
+    const errorCount = this._issues.filter(i => i.severity === 'error').length;
+    return {
+      totalResults: this._results.length,
+      totalIssues: this._issues.length,
+      errorCount,
+      warningCount: this._issues.filter(i => i.severity === 'warning').length,
+      infoCount: this._issues.filter(i => i.severity === 'info').length,
+      valid: errorCount === 0,
+    };
+  }
+
+  getFailures(): ValidationResult[] {
+    return this._results
+      .filter(r => r.issues.some(i => i.severity === 'error'))
+      .map(r => ({ valid: r.valid, issues: [...r.issues] }));
+  }
+
+  getIssuesByField(field: string): ValidationIssue[] {
+    return this._issues.filter(i => i.field === field).map(i => ({ ...i }));
+  }
+
+  getIssuesBySeverity(severity: ValidationSeverity): ValidationIssue[] {
+    return this._issues.filter(i => i.severity === severity).map(i => ({ ...i }));
+  }
+
+  getAllIssues(): ValidationIssue[] {
+    return this._issues.map(i => ({ ...i }));
+  }
+}
